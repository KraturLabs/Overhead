cmake_minimum_required(VERSION 3.20)
project(NameplateLab LANGUAGES CXX)
if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/src/native_signatures.h")
  message(FATAL_ERROR "Missing local native contracts. Follow README.md and run tools/prepare_profile.py first.")
endif()
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "Use 32-bit MSVC: configure with -A Win32.")
endif()
if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/dependencies/sdk/Ashita.h")
  message(FATAL_ERROR "Missing SDK. Run prepare-sdk.ps1 first.")
endif()
add_compile_options(/utf-8)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(nameplate_layout STATIC src/layout.cpp src/health.cpp src/options.cpp src/levels.cpp src/traits.cpp src/debuffs.cpp src/actions.cpp)
target_include_directories(nameplate_layout PUBLIC src)
target_compile_options(nameplate_layout PRIVATE /W4 /WX /fp:strict /arch:SSE2)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/layout_runner.cpp")
  add_executable(layout_runner tests/layout_runner.cpp)
  target_link_libraries(layout_runner PRIVATE nameplate_layout)
endif()
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/health_tests.cpp")
  add_executable(health_tests tests/health_tests.cpp)
  target_link_libraries(health_tests PRIVATE nameplate_layout)
  target_compile_options(health_tests PRIVATE /W4 /WX /fp:strict /arch:SSE2)
endif()
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/options_tests.cpp")
  add_executable(options_tests tests/options_tests.cpp)
  target_link_libraries(options_tests PRIVATE nameplate_layout)
  target_compile_options(options_tests PRIVATE /W4 /WX /fp:strict /arch:SSE2)
endif()
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/src/plugin.cpp")
  add_library(nameplatelab SHARED src/plugin.cpp src/nameplatelab.def src/third_party.rc)
  target_link_libraries(nameplatelab PRIVATE nameplate_layout)
  target_include_directories(nameplatelab SYSTEM PRIVATE dependencies/sdk)
  target_compile_options(nameplatelab PRIVATE /W4 /WX /fp:strict /arch:SSE2 /guard:cf-)
  target_link_options(nameplatelab PRIVATE /SAFESEH:NO /Brepro "/MAP:${CMAKE_CURRENT_BINARY_DIR}/nameplatelab.map")
endif()
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/runtime_tests.cpp")
  add_executable(runtime_tests tests/runtime_tests.cpp)
  target_include_directories(runtime_tests SYSTEM PRIVATE dependencies/sdk)
  target_link_libraries(runtime_tests PRIVATE nameplate_layout)
  target_compile_options(runtime_tests PRIVATE /W4 /WX /fp:strict /arch:SSE2 /guard:cf-)
endif()

if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/plugin_lifecycle_tests.cpp")
  add_library(synthetic_client SHARED tests/synthetic_client.cpp)
  target_link_options(synthetic_client PRIVATE /NODEFAULTLIB "/ENTRY:DllMain@12")
  set_target_properties(synthetic_client PROPERTIES OUTPUT_NAME FFXiMain RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_CURRENT_BINARY_DIR}/fixtures")
  add_executable(plugin_lifecycle_tests tests/plugin_lifecycle_tests.cpp)
  target_include_directories(plugin_lifecycle_tests SYSTEM PRIVATE dependencies/sdk)
  target_compile_options(plugin_lifecycle_tests PRIVATE /W4 /WX /fp:strict /arch:SSE2)
endif()

if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/native_discovery_tests.cpp")
  add_executable(native_discovery_tests tests/native_discovery_tests.cpp)
endif()
