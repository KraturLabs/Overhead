cmake_minimum_required(VERSION 3.20)
project(Overhead LANGUAGES CXX)
if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/src/native_signatures.h")
  message(FATAL_ERROR "Missing local native contracts. Follow README.md and run tools/prepare_profile.py first.")
endif()
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "Use 32-bit MSVC: configure with -A Win32.")
endif()
if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/dependencies/sdk/Ashita.h")
  message(FATAL_ERROR "Missing SDK. Run prepare-sdk.ps1 first.")
endif()
add_compile_options(/utf-8)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(nameplate_layout STATIC src/layout.cpp src/text_font.cpp src/health.cpp src/options.cpp src/levels.cpp src/traits.cpp src/debuffs.cpp src/actions.cpp src/xp.cpp src/arrow.cpp)
target_link_libraries(nameplate_layout PUBLIC gdi32)
target_include_directories(nameplate_layout PUBLIC src)
target_compile_options(nameplate_layout PRIVATE /W4 /WX /fp:strict /arch:SSE2)

add_library(overhead SHARED src/plugin.cpp src/overhead.def src/third_party.rc)
target_link_libraries(overhead PRIVATE nameplate_layout)
target_include_directories(overhead SYSTEM PRIVATE dependencies/sdk)
target_compile_options(overhead PRIVATE /W4 /WX /fp:strict /arch:SSE2 /guard:cf-)
target_link_options(overhead PRIVATE /SAFESEH:NO /Brepro "/MAP:${CMAKE_CURRENT_BINARY_DIR}/overhead.map")

if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/targets.cmake")
  include(tests/targets.cmake)
endif()
