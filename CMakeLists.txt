cmake_minimum_required(VERSION 3.20)
project(NameplateLab LANGUAGES CXX)
if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/src/client_profile.h")
  message(FATAL_ERROR "Missing local client profile. Follow README.md and run tools/prepare_profile.py first.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(nameplate_layout STATIC src/layout.cpp src/health.cpp src/options.cpp)
target_include_directories(nameplate_layout PUBLIC src)
target_compile_options(nameplate_layout PRIVATE /W4 /WX /fp:strict /arch:SSE2)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/layout_runner.cpp")
  add_executable(layout_runner tests/layout_runner.cpp)
  target_link_libraries(layout_runner PRIVATE nameplate_layout)
endif()
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/health_tests.cpp")
  add_executable(health_tests tests/health_tests.cpp)
  target_link_libraries(health_tests PRIVATE nameplate_layout)
  target_compile_options(health_tests PRIVATE /W4 /WX /fp:strict /arch:SSE2)
endif()
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/options_tests.cpp")
  add_executable(options_tests tests/options_tests.cpp)
  target_link_libraries(options_tests PRIVATE nameplate_layout)
  target_compile_options(options_tests PRIVATE /W4 /WX /fp:strict /arch:SSE2)
endif()
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/src/plugin.cpp")
  configure_file(src/runtime_path.h.in generated/runtime_path.h @ONLY)
  add_library(nameplatelab SHARED src/loader.cpp src/nameplatelab.def)
  target_include_directories(nameplatelab PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
  add_library(nameplatelab_engine SHARED src/plugin.cpp src/engine.def)
  target_include_directories(nameplatelab_engine SYSTEM PRIVATE dependencies/sdk)
  target_link_libraries(nameplatelab_engine PRIVATE nameplate_layout)
  target_compile_options(nameplatelab_engine PRIVATE /W4 /WX /fp:strict /arch:SSE2 /guard:cf-)
  target_link_options(nameplatelab_engine PRIVATE /SAFESEH:NO /Brepro "/MAP:${CMAKE_CURRENT_BINARY_DIR}/nameplatelab_engine.map")
  target_include_directories(nameplatelab SYSTEM PRIVATE dependencies/sdk)
  target_compile_options(nameplatelab PRIVATE /W4 /WX /fp:strict /arch:SSE2 /guard:cf-)
  target_link_options(nameplatelab PRIVATE /SAFESEH:NO /Brepro)
  target_link_options(nameplatelab PRIVATE "/MAP:${CMAKE_CURRENT_BINARY_DIR}/nameplatelab.map")
endif()
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/runtime_tests.cpp")
  add_executable(runtime_tests tests/runtime_tests.cpp)
  target_include_directories(runtime_tests SYSTEM PRIVATE dependencies/sdk)
  target_link_libraries(runtime_tests PRIVATE nameplate_layout)
  target_compile_options(runtime_tests PRIVATE /W4 /WX /fp:strict /arch:SSE2 /guard:cf-)
endif()

if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/reload_tests.cpp")
  foreach(id RANGE 1 3)
    add_library(reload_fixture_${id} SHARED tests/reload_fixture.cpp tests/reload_fixture.def)
    target_include_directories(reload_fixture_${id} SYSTEM PRIVATE dependencies/sdk)
    if(id EQUAL 3)
      target_compile_definitions(reload_fixture_${id} PRIVATE FIXTURE_ID=${id} FIXTURE_ABI=999)
    else()
      target_compile_definitions(reload_fixture_${id} PRIVATE FIXTURE_ID=${id} FIXTURE_ABI=2)
    endif()
    target_compile_options(reload_fixture_${id} PRIVATE /W4 /WX)
  endforeach()
  add_executable(reload_tests tests/reload_tests.cpp)
  target_include_directories(reload_tests SYSTEM PRIVATE dependencies/sdk)
  target_include_directories(reload_tests PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
  target_compile_options(reload_tests PRIVATE /W4 /WX /fp:strict /arch:SSE2 /guard:cf-)
endif()

if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/engine_lifecycle_tests.cpp")
  add_library(synthetic_client SHARED tests/synthetic_client.cpp)
  target_link_options(synthetic_client PRIVATE /NODEFAULTLIB "/ENTRY:DllMain@12")
  set_target_properties(synthetic_client PROPERTIES OUTPUT_NAME FFXiMain RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_CURRENT_BINARY_DIR}/fixtures")
  add_executable(engine_lifecycle_tests tests/engine_lifecycle_tests.cpp)
  target_include_directories(engine_lifecycle_tests SYSTEM PRIVATE dependencies/sdk)
  target_include_directories(engine_lifecycle_tests PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
  target_compile_options(engine_lifecycle_tests PRIVATE /W4 /WX /fp:strict /arch:SSE2 /guard:cf-)
endif()
